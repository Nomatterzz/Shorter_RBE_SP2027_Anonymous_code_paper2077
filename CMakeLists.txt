cmake_minimum_required(VERSION 3.20)
project(shorter_rbe VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(RBE_SANITIZE "Address and undefined-behaviour sanitizers" OFF)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
find_path(BOOST_INCLUDE_DIR boost/multiprecision/cpp_int.hpp REQUIRED)
add_library(rbe src/arithmetic.cpp src/sampling.cpp src/scheme.cpp src/serialization.cpp)
target_include_directories(rbe PUBLIC include ${BOOST_INCLUDE_DIR})
target_link_libraries(rbe PUBLIC OpenSSL::Crypto)
target_compile_options(rbe PRIVATE -Wall -Wextra -Wconversion -Wshadow)
if(RBE_SANITIZE)
 target_compile_options(rbe PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(rbe PUBLIC -fsanitize=address,undefined)
endif()
add_executable(rbe_cli src/main.cpp)
target_link_libraries(rbe_cli PRIVATE rbe)
add_executable(rbe_tests tests/tests.cpp)
target_link_libraries(rbe_tests PRIVATE rbe)
enable_testing()
add_test(NAME unit_and_registry COMMAND rbe_tests --quick)
add_test(NAME main_parameters COMMAND rbe_tests --main)
add_test(NAME retuned_controls COMMAND rbe_tests --controls)
set_tests_properties(main_parameters retuned_controls PROPERTIES LABELS full TIMEOUT 900)
install(TARGETS rbe_cli RUNTIME DESTINATION bin)
